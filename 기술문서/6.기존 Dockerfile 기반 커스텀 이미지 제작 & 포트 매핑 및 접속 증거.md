# 도커 파일 
가벼운 웹 서버 Nginx 를 가져옵니다.
FROM nginx:alpine

방금 만든 HTML 파일을 도커 안의 웹 서버 폴더로 복사합니다.
COPY index.html /usr/share/nginx/html/index.html

웹 사이트에서 사용할 포트
EXPOSE 80

# index.html 파일
<!DOCTYPE html>
<html>
<head>
    <meta charset="UTF-8">
    <title>나의 도커 웹서버</title>
</head>
<body>
    <h1>오늘의 명언</h1>
    <p>오늘 하루 어제보다 1%라도 성장하면 1년 뒤는 37%를 성장 할 수 있다</p>
    <p>fighting!!!!!</p>
</body>
</html>

# 사용한 베이스
# FROM nginx:alpine
nginx중에서도 alpine를 사용한 이유는 단순한 정적 웹페이지를 서비스하는 목적이므로, 무거운 OS 환경이 포함된 기본 nginx 이미지 대신 초경량 리눅스인 Alpine 기반의 nginx:alpine을 선택했습니다

# dokerfile에서의 커스텀
# EXPOSE 80
웹 사이트에서 사용할 포트를 설명해 줌으로서 나중에 다른사람이 내가 사용한 포트를 알기쉽게 하기 위함입니다


# index.html에서의 커스텀
index.html을 이미지 안에 넣은 이유: 기본 웹페이지 대신 내가 만든 웹페이지를 띄우기 위해서입니다.<br>
또한 저만의 명언을 사이트에 보여줌으로써 동기부여를 부각시키기 위함입니다<br>
또한 8080포트와 8081번 포트를 나누어 하나는 마운트를 적용(8081)시킨 뒤 실시간 변화를 알아보고 비교 실험하였습니다.<br>
그결과 마운트를 적용한 것은 수정 후 새로고침을 하면 즉각적인 변화가 일어나지만 기존의 것은 수정 후 빌드를 하고 실행 후 새로고침을 해야 변화가 일어난다는 사실을 통해 마운트는 개발 과정에서 유용하게 사용할 수 있고 기존의 것은 새로운 버전으로 업데이트 시킬때 좋을 것 같다는 생각이 들었습니다.
