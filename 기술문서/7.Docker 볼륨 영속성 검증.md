# 컨테이너 만들고 접속하기
%docker run -it --name c1 -v v1:/data ubuntu bash

# 컨테이너 안에서 파일 만들고 확인하기
%echo "fighting!" > /data/volume.txt
%cat /data/volume.txt
fighting!

# 컨테이너에서 빠져나오기
%exit
exit

# 컨테이너 삭제하기
%docker rm c1
c1
# 새로운 컨테이너를 띄워서 데이터가 살아있는지 확인하기
%docker run --name c2 -v v1:/data ubuntu cat /data/volume.txt
fighting!
